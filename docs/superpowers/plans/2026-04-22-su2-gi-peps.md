# src/vmc/peps/su2_gi/group.py
"""SU(2) gauge group: CG coefficients, {6j} symbols, intertwiner basis, block tables."""
from __future__ import annotations

from typing import Protocol, runtime_checkable

import jax
import jax.numpy as jnp
import numpy as np


@runtime_checkable
class GaugeGroup(Protocol):
    name: str
    j_max: float | int

    def irreps(self) -> tuple[float, ...]: ...
    def dim(self, j: float) -> int: ...
    def dual(self, j: float) -> float: ...
    def fuse(self, j1: float, j2: float) -> tuple[tuple[float, int], ...]: ...
    def cg(self, j1: float, j2: float, j: float) -> jax.Array: ...
    def casimir(self, j: float) -> float: ...


class SU2:
    """SU(2) gauge group truncated at j_max.

    Condon-Shortley phase convention throughout. CG coefficients computed
    via Racah's formula at construction and cached.

    Magnetic quantum number indexing: m = j, j-1, ..., -j
    maps to array index i = 0, 1, ..., 2j via i = j - m.
    """

    name: str = "SU2"

    def __init__(self, j_max: float) -> None:
        self.j_max = j_max
        self._cg_cache: dict[tuple[float, float, float], jnp.ndarray] = {}
        self._precompute_cgs()

    def irreps(self) -> tuple[float, ...]:
        return tuple(j / 2 for j in range(int(2 * self.j_max) + 1))

    def dim(self, j: float) -> int:
        return int(2 * j + 1)

    def dual(self, j: float) -> float:
        return j  # SU(2) is self-conjugate

    def casimir(self, j: float) -> float:
        return j * (j + 1)

    def fuse(self, j1: float, j2: float) -> tuple[tuple[float, int], ...]:
        results = []
        j_min = abs(j1 - j2)
        j_max_fused = j1 + j2
        j = j_min
        while j <= min(j_max_fused, self.j_max):
            results.append((j, 1))  # SU(2) has multiplicity 1 at 2-way fusion
            j += 1.0 if (j1 + j2 - j) == int(j1 + j2 - j) else 0.5
            # Step by 1 (integer spacing in triangle rule)
        # Correct iteration:
        results = []
        j = j_min
        while j <= min(j_max_fused, self.j_max):
            results.append((j, 1))
            j += 1.0
        return tuple(results)

    def cg(self, j1: float, j2: float, j: float) -> jax.Array:
        key = (j1, j2, j)
        if key not in self._cg_cache:
            self._cg_cache[key] = self._compute_cg(j1, j2, j)
        return self._cg_cache[key]

    def _precompute_cgs(self) -> None:
        for j1 in self.irreps():
            for j2 in self.irreps():
                for j_out, _ in self.fuse(j1, j2):
                    self.cg(j1, j2, j_out)

    def _compute_cg(self, j1: float, j2: float, j: float) -> jnp.ndarray:
        """Compute CG block via Racah's formula. Shape: (dim(j1), dim(j2), dim(j), 1).

        Uses the standard Racah formula for Clebsch-Gordan coefficients in
        Condon-Shortley convention. The trailing axis is multiplicity (always 1
        for SU(2) 2-way fusion).
        """
        d1, d2, d = self.dim(j1), self.dim(j2), self.dim(j)
        cg = np.zeros((d1, d2, d, 1), dtype=np.complex128)
        for i1 in range(d1):
            m1 = j1 - i1
            for i2 in range(d2):
                m2 = j2 - i2
                m = m1 + m2
                if abs(m) > j:
                    continue
                im = int(j - m)
                cg[i1, i2, im, 0] = _racah_cg(j1, m1, j2, m2, j, m)
        return jnp.asarray(cg)


def _racah_cg(j1: float, m1: float, j2: float, m2: float, j: float, m: float) -> float:
    """Clebsch-Gordan coefficient ⟨j1 m1; j2 m2 | j m⟩ via Racah's formula.

    Condon-Shortley convention. Returns real float (imaginary part vanishes
    for integer/half-integer angular momentum in this convention).
    """
    if m != m1 + m2:
        return 0.0
    if j < abs(j1 - j2) or j > j1 + j2:
        return 0.0

    # Standard Racah formula implementation using factorials.
    # Reference: Varshalovich, Moskalev, Khersonskii, "Quantum Theory of
    # Angular Momentum", Chapter 8.
    from math import factorial, sqrt

    def _f(x):
        return factorial(int(x))

    # Triangle coefficient
    tri = (
        _f(j1 + j2 - j) * _f(j1 - j2 + j) * _f(-j1 + j2 + j)
        / _f(j1 + j2 + j + 1)
    )

    prefactor = sqrt(
        (2 * j + 1)
        * _f(j + m) * _f(j - m)
        * _f(j1 - m1) * _f(j1 + m1)
        * _f(j2 - m2) * _f(j2 + m2)
        * tri
    )

    # Sum over k
    k_min = max(0, int(j2 - j - m1), int(j1 + m2 - j))
    k_max = min(int(j1 + j2 - j), int(j1 - m1), int(j2 + m2))
    total = 0.0
    for k in range(k_min, k_max + 1):
        total += ((-1) ** k) / (
            _f(k)
            * _f(int(j1 + j2 - j - k))
            * _f(int(j1 - m1 - k))
            * _f(int(j2 + m2 - k))
            * _f(int(j - j2 + m1 + k))
            * _f(int(j - j1 - m2 + k))
        )

    return prefactor * total
```

def build_intertwiner_basis(
    self,
    j_l: float, j_u: float, j_r: float, j_d: float,
    Qx: float = 0.0,
    *,
    j_qp: float = 0.0,
) -> list[jnp.ndarray]:
    """Build orthonormal intertwiner basis via canonical s-channel tree (spec §4.2).

    Returns list of arrays, one per intertwiner ι. Each array has shape
    (dim(j_l), dim(j_u), dim(j_r), dim(j_d)) for pure gauge, or
    (dim(j_l), dim(j_u), dim(j_r), dim(j_d), dim(j_qp)) with matter.

    Canonical tree:
      (j_l, j_u) -> j_m via CG
      (j_r*, j_d*) -> j_n via CG
      pure gauge: (j_m, j_n) -> Q_x, so ι = (j_m,) with j_m = j_n
      with matter: (j_m, j_n) -> j_k, (j_k, j_qp) -> Q_x, so ι = (j_m, j_n, j_k)

    Normalization: ∏_e 1/√(2 j_e + 1) per internal edge.
    """
```

The implementation contracts CGs along the tree and normalizes per internal edge. For pure gauge with `Q_x=0`, the constraint `j_m = j_n` is enforced by the fusion `(j_m, j_n) → 0` requiring `j_m = j_n`.

from dataclasses import dataclass

@dataclass(frozen=True)
class BlockTable:
    """Enumeration of Gauss-compatible (p, sector_tuple, ι) triples at a vertex."""
    total_blocks: int
    # block_ids[p][tup_id] -> array of shape (ι_max,) with flat indices
    # into tensors[r][c], sentinel = total_blocks for missing entries.
    block_ids: jnp.ndarray  # shape (phys_dim, n_tup_ids, ι_max)
    # Consumption weights w[p, tup_id, ι]
    weights: jnp.ndarray    # shape (phys_dim, n_tup_ids, ι_max)
    # tup_id -> (j_l, j_u, j_r, j_d) mapping
    tup_to_id: dict[tuple[float, ...], int]
    ι_max: int


def build_block_table(
    group: SU2,
    n_rows: int, n_cols: int,
    r: int, c: int,
    Qx: float = 0.0,
    *,
    phys_dim: int = 1,
    charge_of_site: tuple[float, ...] = (0.0,),
) -> BlockTable:
    """Enumerate all allowed (p, tup, ι) triples for vertex (r, c)."""

Each returned intertwiner gets a sequential flat `block_id`. Store `block_ids[p, tup_id, ι]` as the flat index, with sentinel `total_blocks` for padding.

def build_plaquette_table(
    group: SU2,
) -> dict[tuple[float, ...], list[tuple[tuple[float, ...], complex]]]:
    """Enumerate plaquette outcomes for Û□ + Û†□ (spec §5.4).

    For each input (j1, j2, j3, j4), apply each of the 2⁴ fundamental
    raise/lower patterns from Û□ and Û□†, drop P_{j_max}-violating patterns,
    accumulate CG + {6j} products.

    Returns: {input_tuple: [(output_tuple, amplitude), ...]}
    """
```

The algorithm:
1. For each input tuple of 4 link irreps within `[0, j_max]`:
2. For `Û□`: each link gets ±1/2 from the fundamental rep parallel transporter. Link `ℓ` transitions `j_ℓ → j_ℓ'` where `j_ℓ' ∈ {|j_ℓ - 1/2|, j_ℓ + 1/2}` (triangle rule). Reverse for `Û†□`.
3. Drop outcomes where any `j_ℓ' > j_max`.
4. Amplitude = product of CG factors at each corner × {6j} recoupling.
5. Sum amplitudes for `Û□` and `Û†□` (Hermitian combination).
6. Also compute `Z_table[input_id] = Σ_k |amp_k|²`.

    tensors: list[list[nnx.Param]] = nnx.data()

    def __init__(
        self,
        *,
        rngs: nnx.Rngs,
        config: SU2GIPEPSConfig,
        contraction_strategy: Any,
    ) -> None:
        self.config = config
        self.shape = config.shape
        self.strategy = contraction_strategy
        self.phys_dim = config.phys_dim
        self.charge_of_site = config.charge_of_site
        self.dtype = config.dtype
        n_rows, n_cols = self.shape
        group = config.group

        # Parse Qx into per-vertex values
        qx = np.asarray(config.Qx, dtype=np.float64)
        if qx.ndim == 0:
            qx = np.full(self.shape, qx)
        self.Qx = tuple(tuple(float(q) for q in row) for row in qx.tolist())

        # Build block tables per vertex
        block_tables: list[list[BlockTable]] = []
        tensors: list[list[nnx.Param]] = []
        total_blocks_list: list[int] = []
        for r in range(n_rows):
            bt_row: list[BlockTable] = []
            t_row: list[nnx.Param] = []
            for c in range(n_cols):
                bt = build_block_table(
                    group, n_rows, n_cols, r, c,
                    Qx=self.Qx[r][c],
                    phys_dim=config.phys_dim,
                    charge_of_site=config.charge_of_site,
                )
                bt_row.append(bt)
                total_blocks_list.append(bt.total_blocks)
                # Tensor: (total_blocks, D, D, D, D)
                D = config.D
                t_val = random_tensor(
                    rngs,
                    (bt.total_blocks, D, D, D, D),
                    config.dtype,
                ) / (D ** 2)  # scale 1/√(D⁴)
                # Boost trivial block (all j=0, p=0) for electric vacuum overlap
                if bt.total_blocks > 0:
                    t_val = t_val.at[0].add(
                        0.1 * jnp.ones((D, D, D, D), dtype=config.dtype)
                    )
                t_row.append(nnx.Param(t_val, dtype=config.dtype))
            block_tables.append(bt_row)
            tensors.append(t_row)
        self.tensors = tensors
        self.total_blocks = tuple(total_blocks_list)

        # Build plaquette table
        plaq_table = build_plaquette_table(group)
        z_table = {
            tup: sum(abs(amp) ** 2 for _, amp in outcomes)
            for tup, outcomes in plaq_table.items()
        }

        self.tables = SU2GITables(
            block_tables=block_tables,
            plaquette_table=plaq_table,
            z_table=z_table,
            bmps_bond_blocks=[],  # populated by contraction.py

        num_h = n_rows * (n_cols - 1)
        num_v = (n_rows - 1) * n_cols
        if phys_dim > 1:
            num_sites = n_rows * n_cols
            sites = sample[:num_sites].reshape(shape)
            h_links = sample[num_sites:num_sites + num_h].reshape(n_rows, n_cols - 1)
            v_links = sample[num_sites + num_h:].reshape(n_rows - 1, n_cols)
            return sites, h_links, v_links
        h_links = sample[:num_h].reshape(n_rows, n_cols - 1)
        v_links = sample[num_h:].reshape(n_rows - 1, n_cols)

def _assemble_brick(
    tensor: jax.Array,
    block_table: BlockTable,
    tup_id: int,
    p: int = 0,
) -> jax.Array:
    """Assemble one MPO brick from per-sector blocks (spec §5.1).

    1. Gather ≤ ι_max blocks via block_ids[p, tup_id]
    2. Weight by w[p, tup_id, ι]
    3. Sum over ι → shape (D, D, D, D)
    """
    flat_ids = block_table.block_ids[p, tup_id]          # (ι_max,)
    raw = jnp.take(tensor, flat_ids, axis=0, mode='fill', fill_value=0)  # (ι_max, D, D, D, D)
    weights = block_table.weights[p, tup_id]              # (ι_max,)
    return jnp.einsum("i,ijklm->jklm", weights, raw)

        irreps = jnp.asarray([group.casimir(j) for j in group.irreps()])
        for row, col in self.sites:
            total = total + irreps[links[row, col]]
        return total

    def tree_flatten(self):
        return (self.diag,), (self.sites, self.orientation)

    @classmethod
    def tree_unflatten(cls, aux_data, children):
        (diag,) = children
        sites, orientation = aux_data
        return cls(sites=sites, diag=diag, orientation=orientation)


@jax.tree_util.register_pytree_node_class
@dataclass(frozen=True)
class PlaquetteSU2Term(TransitionOperator):
    """SU(2) plaquette term Û□ + Û†□ (spec §5.4)."""

    row: int
    col: int

    @property
    def sites(self) -> tuple[tuple[int, int], ...]:
        return (
            (self.row, self.col), (self.row, self.col + 1),
            (self.row + 1, self.col), (self.row + 1, self.col + 1),
        )

    def tree_flatten(self):
        return (), (self.row, self.col)

    @classmethod
    def tree_unflatten(cls, aux_data, children):
        del children
        row, col = aux_data
        return cls(row=row, col=col)


@support_span.dispatch
def support_span(_: PlaquetteSU2Term) -> tuple[int, int]:
    return 2, 2


@support_span.dispatch
def support_span(_: LinkCasimirTerm) -> tuple[int, int]:
    return 1, 1

@dataclass(frozen=True)
class HorizontalMatterHoppingSU2Term(TransitionOperator):

    row: int

@dataclass(frozen=True)

    """Gauge-covariant hopping ψ†_x U_{xy} ψ_y on vertical link."""

    row: int
    col: int

    @property
    def sites(self) -> tuple[tuple[int, int], ...]:
        return ((self.row, self.col), (self.row + 1, self.col))

def build_su2_hamiltonian(
    shape: tuple[int, int],
    group,
    *,
    g_E: float = 1.0,
    g_B: float = 1.0,
    phys_dim: int = 1,
    charge_of_site: tuple[float, ...] = (0.0,),
    m_mass: float = 0.0,
    w_hop: float = 0.0,
) -> SU2LocalHamiltonian:
    """Build the SU(2) Kogut-Susskind Hamiltonian (spec §2)."""
    n_rows, n_cols = shape
    terms = []
    coeffs = []

    # Electric terms: g_E * Σ_links j(j+1)
    casimir_diag = jnp.asarray([group.casimir(j) for j in group.irreps()])
    for r in range(n_rows):
        for c in range(n_cols - 1):
            terms.append(LinkCasimirTerm(sites=((r, c),), diag=casimir_diag, orientation="h"))
            coeffs.append(g_E)
    for r in range(n_rows - 1):
        for c in range(n_cols):
            terms.append(LinkCasimirTerm(sites=((r, c),), diag=casimir_diag, orientation="v"))
            coeffs.append(g_E)

    # Plaquette terms: -g_B * Σ_plaq (Û□ + Û†□)
    for r in range(n_rows - 1):
        for c in range(n_cols - 1):
            terms.append(PlaquetteSU2Term(row=r, col=c))
            coeffs.append(-g_B)

    # Matter terms (when phys_dim ≥ 2)
    if phys_dim >= 2:
        charge_diag = jnp.asarray(charge_of_site)
        for r in range(n_rows):
            for c in range(n_cols):
                terms.append(OnSiteMassSU2Term(sites=((r, c),), diag=charge_diag))
                coeffs.append(m_mass)
        for r in range(n_rows):
            for c in range(n_cols - 1):
                terms.append(HorizontalMatterHoppingSU2Term(row=r, col=c))
                coeffs.append(-w_hop)
        for r in range(n_rows - 1):
            for c in range(n_cols):
                terms.append(VerticalMatterHoppingSU2Term(row=r, col=c))
                coeffs.append(-w_hop)

    return SU2LocalHamiltonian(
        shape=shape,
        terms=tuple(terms),
        coeffs=tuple(jnp.asarray(c) for c in coeffs),
    )

    """Build SU(2) GI-PEPS init_cache/transition/estimate kernels (spec §5.6)."""
    shape = model.shape

    )
    has_time_dep = any(s is not None for s in coeff_structure.schedules)
    static_coeffs = None if has_time_dep else coeff_structure.build_coeffs()

    def init_cache(tensors, config_states, t=None):
        def build_one(config_state):
            sites_val, h_links, v_links = SU2GIPEPS.unflatten_sample(
                config_state, shape, phys_dim=config.phys_dim

            dtype = tensors[0][0].dtype
